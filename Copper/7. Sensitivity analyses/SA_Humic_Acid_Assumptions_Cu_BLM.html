Objectives

To assess BLM’s assumption of humic acid content (HA) of dissolved organic carbon (DOC), we will assess bioavailability across a range of HA % 10-60 by increments of 10.

3. Sensitivity Analysis

Computed differences in TBL-Cu (nmol/g) within each Sample_ID for each incremental increase in HA (%). To do so, first create ha_sens

library(dplyr)

ha_sens <- ha_dat %>%
  arrange(Sample_ID, HA_content) %>%
  group_by(Sample_ID, Typology) %>%
  
  # Create "next HA" value within each sample
  mutate(
    HA_next = lead(HA_content),
    TBL_next = lead(TBL_Cu_nmol_g)
  ) %>%
  
  # Keep only valid transitions (drop last HA=60 rows)
  filter(!is.na(HA_next)) %>%
  
  # Optional: ensure only 10-step increments
  filter(HA_next - HA_content == 10)

i. Calculate metrics

  • Delta: absolute change

  • Percent change: relative change

  • Log response ratio (LRR): log10(next / current)

ha_sens <- ha_sens %>%
  mutate(
    delta = TBL_next - TBL_Cu_nmol_g,
    
    pct_change = (TBL_next - TBL_Cu_nmol_g) / TBL_Cu_nmol_g * 100,
    
    log_rr = log10(TBL_next / TBL_Cu_nmol_g),
    
    HA_transition = paste0(HA_content, "-", HA_next)
  )

ii. Summary statistics

Mean and SE of each SA metric by typology by incremental increase in HA (%)

ha_summary <- ha_sens %>%
  group_by(Typology, HA_transition) %>%
  summarise(
    mean_delta = mean(delta, na.rm = TRUE),
    se_delta   = sd(delta, na.rm = TRUE) / sqrt(n()),
    
    mean_pct_change = mean(pct_change, na.rm = TRUE),
    se_pct_change   = sd(pct_change, na.rm = TRUE) / sqrt(n()),
    
    mean_log_rr = mean(log_rr, na.rm = TRUE),
    se_log_rr   = sd(log_rr, na.rm = TRUE) / sqrt(n()),
    
    n = n(),
    .groups = "drop"
  )
